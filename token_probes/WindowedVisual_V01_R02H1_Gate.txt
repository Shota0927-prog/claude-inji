// MA Source packet, evaluated in the 1m context (B29 request). EMA 2000 / 3000 of close
// and their raw deltas over `lookback` bars; native na during warmup.
// Returns [maSourceOpenTime, maSourceCloseTime, ema2000, ema2000Slope, ema3000, ema3000Slope].
maPackV2(simple int lookback) =>
    float ema2000Value = ta.ema(close, 2000)
    float ema3000Value = ta.ema(close, 3000)
    float ema2000Slope = ema2000Value - ema2000Value[lookback]
    float ema3000Slope = ema3000Value - ema3000Value[lookback]
    [time, time_close, ema2000Value, ema2000Slope, ema3000Value, ema3000Slope]

// Swing Source event packet, evaluated in the 5m / 15m / 1h context. A pivot confirmed
// on this bar (len bars each side) carries its price, the pivot bar's open / close time
// and this bar's close time; no pivot -> na (an event, never a carried state).
// Returns [highPrice, highOriginTime, highOriginEndTime, highConfirmedTime,
//          lowPrice,  lowOriginTime,  lowOriginEndTime,  lowConfirmedTime].
pivotPackV2(simple int len) =>
    float pivotHigh = ta.pivothigh(high, len, len)
    float pivotLow  = ta.pivotlow(low, len, len)
    bool  hasHigh   = not na(pivotHigh)
    bool  hasLow    = not na(pivotLow)
    float highPrice         = hasHigh ? pivotHigh : na
    int   highOriginTime    = hasHigh ? time[len] : int(na)
    int   highOriginEndTime = hasHigh ? time_close[len] : int(na)
    int   highConfirmedTime = hasHigh ? time_close : int(na)
    float lowPrice          = hasLow ? pivotLow : na
    int   lowOriginTime     = hasLow ? time[len] : int(na)
    int   lowOriginEndTime  = hasLow ? time_close[len] : int(na)
    int   lowConfirmedTime  = hasLow ? time_close : int(na)
    [highPrice, highOriginTime, highOriginEndTime, highConfirmedTime, lowPrice, lowOriginTime, lowOriginEndTime, lowConfirmedTime]

// Accum condition on this Source bar. The window is the last rangeLen bodies (open /
// close ends, never wicks); one rangeMid for the whole window. Native na / history
// semantics; no fill, no custom warmup. Returns [isAccum, rangeHigh, rangeLow].
accumCondV2(simple int rangeLen, simple int baseLen, simple int atrLen, simple int minUpper, simple int minLower, simple int maxRun, float atrMult, float barRatioMax, float driftMax) =>
    float bodyTop   = math.max(open, close)
    float bodyBot   = math.min(open, close)
    float rangeHigh = ta.highest(bodyTop, rangeLen)
    float rangeLow  = ta.lowest(bodyBot, rangeLen)
    float rangeNow  = rangeHigh - rangeLow
    float rangeMid  = (rangeHigh + rangeLow) * 0.5
    float rangeBase = ta.highest(bodyTop, baseLen) - ta.lowest(bodyBot, baseLen)
    float atrValue  = ta.atr(atrLen)
    int upperCloses = 0
    int lowerCloses = 0
    int bullRun     = 0
    int bearRun     = 0
    int maxBullRun  = 0
    int maxBearRun  = 0
    for i = 0 to rangeLen - 1
        if close[i] > rangeMid
            upperCloses += 1
        else if close[i] < rangeMid
            lowerCloses += 1
        if close[i] > open[i]
            bullRun += 1
            bearRun := 0
        else if close[i] < open[i]
            bearRun += 1
            bullRun := 0
        else
            bullRun := 0
            bearRun := 0
        maxBullRun := math.max(maxBullRun, bullRun)
        maxBearRun := math.max(maxBearRun, bearRun)
    float startMid = (open[rangeLen - 1] + close[rangeLen - 1]) * 0.5
    float endMid   = (open + close) * 0.5
    float drift    = math.abs(endMid - startMid)
    bool condAtr      = atrValue > 0 and rangeNow <= atrValue * atrMult
    bool condBar      = rangeBase > 0 and rangeNow <= rangeBase * barRatioMax
    bool condBalance  = upperCloses >= minUpper and lowerCloses >= minLower
    bool condDrift    = rangeNow > 0 and drift / rangeNow <= driftMax
    bool condColorRun = maxBullRun <= maxRun and maxBearRun <= maxRun
    bool isAccum = condAtr and condBar and condBalance and condDrift and condColorRun
    [isAccum, rangeHigh, rangeLow]

// Accum Source packet (1H / 4H / D context, B29). A Candidate state machine local to
// this helper (not a Root): false -> true starts it on the window's body range, true ->
// true widens it with the current body, true -> false emits the Candidate once as a
// Confirmed Box (the ending bar is not part of it) and resets. Event values are this
// bar only. Returns [newConfirmedBox, boxTop, boxBottom, boxId, candidateStartTime,
// confirmedTime, candidateActive, candidateTop, candidateBottom].
accumPackV2(simple int rangeLen, simple int baseLen, simple int atrLen, simple int minUpper, simple int minLower, simple int maxRun, float atrMult, float barRatioMax, float driftMax) =>
    [isAccum, rangeHigh, rangeLow] = accumCondV2(rangeLen, baseLen, atrLen, minUpper, minLower, maxRun, atrMult, barRatioMax, driftMax)
    var bool  candidateActive    = false
    var float candidateTop       = na
    var float candidateBottom    = na
    var int   candidateStartTime = na
    bool  newConfirmedBox       = false
    float boxTop                = na
    float boxBottom             = na
    int   boxId                 = 0
    int   confirmedTime         = na
    int   candidateStartTimeOut = na
    if isAccum and not candidateActive
        candidateActive    := true
        candidateStartTime := time
        candidateTop       := rangeHigh
        candidateBottom    := rangeLow
        candidateStartTimeOut := candidateStartTime
    else if isAccum and candidateActive
        float currentBodyTop    = math.max(open, close)
        float currentBodyBottom = math.min(open, close)
        candidateTop    := math.max(candidateTop, currentBodyTop)
        candidateBottom := math.min(candidateBottom, currentBodyBottom)
        candidateStartTimeOut := candidateStartTime
    else if not isAccum and candidateActive
        newConfirmedBox       := true
        boxTop                := candidateTop
        boxBottom             := candidateBottom
        boxId                 := candidateStartTime
        confirmedTime         := time_close
        candidateStartTimeOut := candidateStartTime
        candidateActive    := false
        candidateTop       := na
        candidateBottom    := na
        candidateStartTime := na
    [newConfirmedBox, boxTop, boxBottom, boxId, candidateStartTimeOut, confirmedTime, candidateActive, candidateTop, candidateBottom]

// FVG Source fact packet (1H / 4H / D context, B29). This bar is the third bar: Bull
// low > high[2] (NativeRange high[2] -> low), Bear high < low[2] (NativeRange high ->
// low[2]), strict; no event -> na ranges. Fresh and structural invalidation belong to
// the Production. Returns [srcOpenTime, srcCloseTime, bullNew, bullBottom, bullTop,
// bearNew, bearBottom, bearTop, high, low, close].
fvgPackV2() =>
    bool  bullNew    = low > high[2]
    float bullBottom = bullNew ? high[2] : na
    float bullTop    = bullNew ? low : na
    bool  bearNew    = high < low[2]
    float bearBottom = bearNew ? high : na
    float bearTop    = bearNew ? low[2] : na
    [time, time_close, bullNew, bullBottom, bullTop, bearNew, bearBottom, bearTop, high, low, close]

// Visual V01 (display only, barstate.islast only). Reads the facade rows; writes nothing back.

// Production code values (Main SIDE_* / PH_* / GR_*), mirrored for display decoding only
const int V_SUPPORT    = 1
const int V_RESISTANCE = -1
const int V_WAITING    = 0
const int V_ARMED      = 1
const int V_ACTIVE     = 2
const int V_BROKEN     = 3
const int V_FLIP_WAIT  = 4
const int V_DORMANT    = 5
const int V_STRONG     = 0
const int V_NEUTRAL    = 1
const int V_WEAK       = 2
const int V_UNAVAIL    = 3
const int POOL_SIZE    = 30

// visual filter on Production fields only (eligible / side / grade / phase / finite range)
visPass(ZE.VisualZoneRow v) =>
    bool sideOk  = v.side == V_SUPPORT ? showSupport : v.side == V_RESISTANCE ? showResistance : false
    bool gradeOk = v.grade == V_STRONG ? showStrong : v.grade == V_NEUTRAL ? showNeutral : v.grade == V_WEAK ? showWeak : v.grade == V_UNAVAIL ? showUnavailable : false
    bool phaseOk = v.phase == V_WAITING ? showWaiting : v.phase == V_ARMED ? showArmed : v.phase == V_ACTIVE ? showActiveTouch : v.phase == V_BROKEN ? showBroken : v.phase == V_FLIP_WAIT ? showFlipWait : v.phase == V_DORMANT ? showDormant : false
    bool rangeOk = not na(v.effectiveBottom) and not na(v.effectiveTop) and v.effectiveBottom <= v.effectiveTop
    (v.eligible or not eligibleOnly) and sideOk and gradeOk and phaseOk and rangeOk

// distance of the display price to [bottom, top]: 0 inside
visDist(float bottom, float top, float px) =>
    px < bottom ? bottom - px : px > top ? px - top : 0.0

// strict order: distance ASC, coreId ASC, generationId ASC, Support before Resistance
visBefore(float d1, int c1, int g1, int s1, float d2, int c2, int g2, int s2) =>
    d1 < d2 ? true : d1 > d2 ? false : c1 < c2 ? true : c1 > c2 ? false : g1 < g2 ? true : g1 > g2 ? false : s1 > s2

visTransp(int grade) =>
    grade == V_STRONG ? transpStrong : grade == V_NEUTRAL ? transpNeutral : grade == V_WEAK ? transpWeak : transpUnavail

// fixed object pools: 30 boxes + 30 lines, created lazily on the first barstate.islast execution only (none on historical
// bars), then reused through setters; never deleted / recreated. The debugPerf table follows the same rule.
var array<box>  boxPool   = array.new_box(0)
var array<line> linePool  = array.new_line(0)
var table       perfTable = table(na)

if barstate.islast
    if array.size(boxPool) == 0
        for p = 0 to POOL_SIZE - 1
            array.push(boxPool, box.new(bar_index, close, bar_index, close, border_color = color(na), bgcolor = color(na)))
            array.push(linePool, line.new(bar_index, close, bar_index, close, color = color(na)))
    if debugPerf and na(perfTable)
        perfTable := table.new(position.top_right, 2, 6)
    // bounded top-K nearest selection: one scan of the rows, insertion into <= maxZonesDrawn slots holding the drawn fields
    int n = array.size(rows)
    array<float> kDist  = array.new_float(0)
    array<int>   kCore  = array.new_int(0)
    array<int>   kGen   = array.new_int(0)
    array<int>   kSide  = array.new_int(0)
    array<float> kBot   = array.new_float(0)
    array<float> kTop   = array.new_float(0)
    array<int>   kGrade = array.new_int(0)
    array<int>   kPhase = array.new_int(0)
    int eligibleCount = 0
    int passCount     = 0
    float px = close
    for i = 0 to n - 1
        if i >= n
            break
        ZE.VisualZoneRow v = array.get(rows, i)
        if v.eligible
            eligibleCount += 1
        if visPass(v)
            passCount += 1
            float d = visDist(v.effectiveBottom, v.effectiveTop, px)
            int pos = array.size(kDist)
            while pos > 0 and visBefore(d, v.coreId, v.generationId, v.side, array.get(kDist, pos - 1), array.get(kCore, pos - 1), array.get(kGen, pos - 1), array.get(kSide, pos - 1))
                pos -= 1
            if pos < maxZonesDrawn
                array.insert(kDist, pos, d)
                array.insert(kCore, pos, v.coreId)
                array.insert(kGen, pos, v.generationId)
                array.insert(kSide, pos, v.side)
                array.insert(kBot, pos, v.effectiveBottom)
                array.insert(kTop, pos, v.effectiveTop)
                array.insert(kGrade, pos, v.grade)
                array.insert(kPhase, pos, v.phase)
                if array.size(kDist) > maxZonesDrawn
                    array.pop(kDist)
                    array.pop(kCore)
                    array.pop(kGen)
                    array.pop(kSide)
                    array.pop(kBot)
                    array.pop(kTop)
                    array.pop(kGrade)
                    array.pop(kPhase)
    // draw: range -> box, point (bottom == top) -> line; unused pool objects are made transparent
    int x1 = bar_index - drawBarsLeft
    int x2 = bar_index + drawBarsRight
    int boxUsed  = 0
    int lineUsed = 0
    int kN = array.size(kDist)
    for j = 0 to kN - 1
        if j >= kN
            break
        float bot    = array.get(kBot, j)
        float top    = array.get(kTop, j)
        int   phase  = array.get(kPhase, j)
        color baseC  = array.get(kSide, j) == V_SUPPORT ? supportColor : resistanceColor
        int   tr     = visTransp(array.get(kGrade, j))
        color fillC  = color.new(baseC, tr)
        color edgeC  = color.new(baseC, math.max(tr - 30, 0))
        string styleV = phase == V_ARMED ? line.style_dashed : line.style_solid
        int   widthV = phase == V_ACTIVE ? 2 : 1
        if bot < top
            box b = array.get(boxPool, boxUsed)
            box.set_lefttop(b, x1, top)
            box.set_rightbottom(b, x2, bot)
            box.set_bgcolor(b, fillC)
            box.set_border_color(b, edgeC)
            box.set_border_style(b, styleV)
            box.set_border_width(b, widthV)
            boxUsed += 1
        else
            line l = array.get(linePool, lineUsed)
            line.set_xy1(l, x1, bot)
            line.set_xy2(l, x2, bot)
            line.set_color(l, edgeC)
            line.set_style(l, styleV)
            line.set_width(l, widthV)
            lineUsed += 1
    for j = boxUsed to POOL_SIZE - 1
        if j >= POOL_SIZE
            break
        box b = array.get(boxPool, j)
        box.set_bgcolor(b, color(na))
        box.set_border_color(b, color(na))
    for j = lineUsed to POOL_SIZE - 1
        if j >= POOL_SIZE
            break
        line.set_color(array.get(linePool, j), color(na))
    if debugPerf
        table.cell(perfTable, 0, 0, "engineWindowDays")
        table.cell(perfTable, 1, 0, str.tostring(engineWindowDays))
        table.cell(perfTable, 0, 1, "row count")
        table.cell(perfTable, 1, 1, str.tostring(n))
        table.cell(perfTable, 0, 2, "filtered")
        table.cell(perfTable, 1, 2, str.tostring(passCount))
        table.cell(perfTable, 0, 3, "drawn")
        table.cell(perfTable, 1, 3, str.tostring(kN))
        table.cell(perfTable, 0, 4, "box used")
        table.cell(perfTable, 1, 4, str.tostring(boxUsed))
        table.cell(perfTable, 0, 5, "line used")
        table.cell(perfTable, 1, 5, str.tostring(lineUsed))

plot(close, title = "Compile anchor", display = display.none)
