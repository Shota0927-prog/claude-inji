f_satBand() =>
    // --- 共通 Basis（SMA でも EMA でもなく VWMA。source は hlc3、length は 5） ---
    float basisV  = ta.vwma(hlc3, 5)
    float commonV = basisV * 0.5

    // --- ATR（ta.atr() なので内部は RMA。現在足ではなく 1 本前の値） ---
    float atr50V = ta.atr(50)[1]

    // --- Anchor 共通 source と State 反転判定用の値（いずれも 1 本前の足から） ---
    float srcV      = close[1] * 0.5
    float trigLowV  = low[1]   * 0.5
    float trigHighV = high[1]  * 0.5

    bool readyV = not na(atr50V) and not na(srcV) and not na(trigLowV) and not na(trigHighV)

    // --- LINE1（k = 5.0） ---
    //   同一バー内で更新済みの anchor / dir を条件判定へ再利用しないよう、
    //   判定前に prevAnchor / prevDir へ退避してから State を更新する。
    float k1V = 5.0
    var float anchor1V = na
    var int   dir1V    = 1
    float prevAnchor1V = anchor1V
    int   prevDir1V    = dir1V
    if readyV
        if na(prevAnchor1V)
            // 初期化：Bull から開始する
            anchor1V := srcV - k1V * atr50V
            dir1V    := 1
        else if prevDir1V == 1
            if trigLowV < prevAnchor1V
                // Bull → Bear
                dir1V    := -1
                anchor1V := srcV + k1V * atr50V
            else
                dir1V    := 1
                anchor1V := math.max(prevAnchor1V, srcV - k1V * atr50V)
        else
            if trigHighV > prevAnchor1V
                // Bear → Bull
                dir1V    := 1
                anchor1V := srcV - k1V * atr50V
            else
                dir1V    := -1
                anchor1V := math.min(prevAnchor1V, srcV + k1V * atr50V)
    float line1V = commonV + anchor1V

    // --- LINE2（k = 2.5。構造は LINE1 と完全に同じで、違うのは倍率だけ） ---
    float k2V = 2.5
    var float anchor2V = na
    var int   dir2V    = 1
    float prevAnchor2V = anchor2V
    int   prevDir2V    = dir2V
    if readyV
        if na(prevAnchor2V)
            // 初期化：Bull から開始する
            anchor2V := srcV - k2V * atr50V
            dir2V    := 1
        else if prevDir2V == 1
            if trigLowV < prevAnchor2V
                // Bull → Bear
                dir2V    := -1
                anchor2V := srcV + k2V * atr50V
            else
                dir2V    := 1
                anchor2V := math.max(prevAnchor2V, srcV - k2V * atr50V)
        else
            if trigHighV > prevAnchor2V
                // Bear → Bull
                dir2V    := 1
                anchor2V := srcV - k2V * atr50V
            else
                dir2V    := -1
                anchor2V := math.min(prevAnchor2V, srcV + k2V * atr50V)
    float line2V = commonV + anchor2V

    // --- STB（Donchian 構造。LINE1 / LINE2 の State とは完全に独立） ---
    //   表示用     ： ta.highest(high, 10) / ta.lowest(low, 10)
    //   方向判定用 ： ta.highest(high, 20)[1] / ta.lowest(low, 20)[1]
    //   毎バー high breakout を先に判定し、同じ足で上下両方を突破した場合も
    //   Bull 側（+1）が優先される。どちらも突破していなければ直前の State を維持。
    float stbUpperV = ta.highest(high, 10)
    float stbLowerV = ta.lowest(low, 10)
    float prev20HighV = ta.highest(high, 20)[1]
    float prev20LowV  = ta.lowest(low, 20)[1]
    var int stbDirV = 1
    if not na(prev20HighV) and not na(prev20LowV)
        if high > prev20HighV
            stbDirV := 1
        else if low < prev20LowV
            stbDirV := -1
    // UP は下側（lower10）、DOWN は上側（upper10）を表示価格にする
    float stbV = stbDirV == 1 ? stbLowerV : stbUpperV

    // --- 青バンド / 赤バンド ---
    //   現在価格と LINE1 / LINE2 / STB の 3 本の位置関係だけで決める。
    //   各ラインの dir は判定に使わない。
    int stateV = 0
    if close > line1V and close > line2V and close > stbV
        stateV := 1
    else if close < line1V and close < line2V and close < stbV
        stateV := -1

    [line1V, line2V, stbV, dir1V, dir2V, stbDirV, stateV, basisV, commonV, atr50V, anchor1V, anchor2V]
